============================================================
ربط المتجر بـ Google Sheets (استقبال الأوردرات في شيت)
============================================================

الخطوة 1: افتح شيت جديد على Google Sheets
   - https://sheets.google.com → جديدة (Blank)
   - في الصف الأول اكتب العناوين دي (كل واحد في عمود):
     التاريخ | رقم الطلب | الاسم | التليفون | العنوان | ملاحظات | التوصيل | الإجمالي | تفاصيل الطلب
   - (ممكن تحذف أي عمود مش محتاجه — الكود بيملا اللي موجود)

الخطوة 2: افتح محرر الكود
   - من القائمة: Extensions (إضافات) ← Apps Script
   - امسح اللي جوه والصق الكود ده:

------------------------------------------------------------
// مفتاح قراءة الأوردرات — غيّره لأي كلمة تحبها، واكتب نفس القيمة
// في اللوحة: إعدادات المتجر ← "مفتاح قراءة الأوردرات"
const ACCESS_KEY = 'minyawi-orders';

function doPost(e) {
  var p = e.parameter;
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];
  sheet.appendRow([
    new Date(),
    p.ref || '',
    p.name || '',
    p.phone || '',
    p.address || '',
    p.notes || '',
    p.delivery || '',
    p.total || '',
    p.lines || ''
  ]);
  // مهم جدًا: عمود التليفون لازم يفضل نص (Text) — لو اتخزن كرقم
  // جوجل شيت بيحذف الـ 0 الأولانية (01012345678 → 1012345678)
  var lastRow = sheet.getLastRow();
  sheet.getRange(2, 4, Math.max(lastRow - 1, 1), 1).setNumberFormat('@');
  sheet.getRange(lastRow, 4).setValue(p.phone || '');
  // مسح الكاش عشان الطلب الجديد يظهر فورًا في اللوحة
  var cache = CacheService.getScriptCache();
  cache.remove('orders:' + (p.k || ACCESS_KEY));
  return ContentService.createTextOutput('OK');
}

function doGet(e) {
  var key = (e.parameter && e.parameter.k) || '';
  if (key !== ACCESS_KEY) {
    return ContentService.createTextOutput(JSON.stringify({ ok: false, error: 'wrong_key' }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  try {
    // نسخة مخزنة (٣٠ ثانية) — عشان قراءة الأوردرات تبقى سريعة
    var cache = CacheService.getScriptCache();
    var cached = cache.get('orders:' + key);
    if (cached) {
      return ContentService.createTextOutput(cached)
        .setMimeType(ContentService.MimeType.JSON);
    }
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];
    var values = sheet.getDataRange().getValues();
    var orders = [];
    for (var i = 1; i < values.length; i++) {
      var r = values[i];
      // استرجاع الـ 0 المحذوفة من الأرقام القديمة المخزنة كأرقام
      // (الموبايل المصري دايمًا 11 رقم: 01x xxxxxxx)
      var ph = String(r[3] || '');
      if (/^1\d{9}$/.test(ph)) ph = '0' + ph;
      orders.push({
        date: r[0] instanceof Date ? r[0].toISOString() : String(r[0] || ''),
        ref: String(r[1] || ''),
        name: String(r[2] || ''),
        phone: ph,
        address: String(r[4] || ''),
        notes: String(r[5] || ''),
        delivery: String(r[6] || ''),
        total: String(r[7] || ''),
        lines: String(r[8] || '')
      });
    }
    orders.reverse();
    if (orders.length > 50) orders = orders.slice(0, 50);
    var out = JSON.stringify({ ok: true, orders: orders });
    cache.put('orders:' + key, out, 30);
    return ContentService.createTextOutput(out)
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ ok: false, error: 'read_failed' }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

الخطوة 3: انشر الكود كـ Web App
   - زرار Deploy (أعلى يمين) ← New deployment
   - Type: Web app
   - Description: (أي حاجة)
   - Execute as: Me (انت)
   - Who has access: Anyone (أي حد — مهم جدًا)
   - زرار Deploy
   - هيطلب منك أذونات: اختار الحساب ← Advanced ← Go to project (unsafe) ← Allow
   - انسخ الرابط اللي هيظهر (اللي ينتهي بـ /exec)

الخطوة 4: حط الرابط في اللوحة
   - افتح لوحة تحكم الموقع (admin.html) ← إعدادات المتجر ← حقل "رابط الشيت"
   - الصق الرابط ← حفظ الإعدادات
   - دوس "تجربة الشيت" — لو ظهر "وصلت الشيت ✅" افتح الشيت وشوف آخر صف

الخطوة 5 (مهمة): فعّل قراءة الأوردرات من اللوحة
   - الكود الجديد فيه doGet (قراءة الأوردرات) + doPost (تسجيل الأوردرات)
   - بعد ما لصقت الكود الجديد، لازم تحدّث النشر: Deploy ← Manage deployments
     ← دوس علامة التعديل (القلم) ← Version: New version ← Deploy
   - الرابط هيبقى نفس الرابط — ما يتغيرش
   - لو الرابط اتغير (لو عملت New deployment جديد) حط الرابط الجديد في اللوحة

الخطوة 6 (مهمة جدًا — السرعة): نزّل نسخة الكود السريع
   - النسخة الحالية فيها كاش (CacheService) عشان قراءة الأوردرات متبقاش بطيئة
   - لو نسخت الكود قبل كده: اعمل copy للكود الجديد ده كله (مع خطوة 5 تحديث الإصدار)
   - أول تحديث بعد خمول هيلاقي تأخير صغير (كولد ستارت من جوجل) — اللي بعده بيبقى فوري

الخطوة 7 (إصلاح رقم العميل الناقص): شغّل نسخة الكود دي
   - المشكلة: جوجل شيت كان بيخزن رقم التليفون كرقم فيحذف الـ 0 الأولانية (010 → 10)
   - الكود الجديد بيخزن التليفون كنص دايما + بيرجع الـ 0 المحذوفة من الطلبات القديمة
   - بعد ما تلصق الكود الجديد: Deploy ← Manage deployments ← القلم
     ← Version: New version ← Deploy (نفس الخطوة 5)
   - افتح شيتك: لو فيه أرقام 10 خانات، حددهم ← Format ← Number ← Plain text
     ← اكتب 0 قدامهم باليد (اللي بعد كده هيتصلح لوحده)

ملحوظات
============================================================
- الأوردرات بتتسجل في Firebase (الموقع نفسه) فورًا — اللوحة بتقرا منها مباشرة
- الشيت بقى نسخة احتياطية فقط (بياخد نفس الطلب مع كل تأكيد)
- لازم يبقى فيه رابط شيت في اللوحة؟ لسه محتاجينه مرتين بس:
  1) الترحيل: أول مرة "نقل الأوردرات من الشيت لـ Firebase" (مرة واحدة)
  2) النسخة الاحتياطية: كل طلب بيتكتب صف في الشيت
- التليجرام والواتساب بيشتغلوا من غير الشيت خالص
- لو الرابط اتحط غلط أو جوجل واقف: مفيش مشكلة — الأوردرات سليمة في Firebase
- قسم "الأوردرات الأخيرة" في اللوحة بيقرا من Firebase (والشيت لو Firebase فاضي)
